        min_x, min_y = -1, -1
        
        for i in range(N):
            for j in range(N):
                if not visited[i][j] and dist[i][j] < min_cost:
                    min_cost = dist[i][j]
                    min_x, min_y = i, j
        
        # 선택한 정점 방문 처리
        visited[min_x][min_y] = True
